	id = dark_world_events.1

	hide_window = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				years_passed >= 50

# Dark World creation 2: GFX event - storm creation
planet_event = {
	id = dark_world_events.3
	title = "dark_world_events.3.name"
	desc = "dark_world_events.3.desc"
	picture = GFX_dark_storm
	show_sound = event_red_alert
	location = this

	hide_window = no
	is_triggered_only = yes

	immediate = {
		add_deposit = d_darkling_storm
		add_planet_devastation = 20
		every_owned_pop_group = {
			limit = {
				NOR = {
					has_trait = trait_mechanical
					has_trait = trait_machine_unit
				}
			}
			kill_pop_group = {
				pop_group = this
				percentage = 0.5
			}
		}
		planet_event = {
			id = dark_world_events.4
			days = 10
		}
	}

    option = {
        name = "We will recover."
    }
}

# Dark World creation 3: GFX event - Ecological disaster!
planet_event = {
	id = dark_world_events.4
	title = "dark_world_events.4.name"
	desc = "dark_world_events.4.desc"
	picture = GFX_ecological_disaster
	show_sound = event_plasma_trails
	location = this

	hide_window = no
	is_triggered_only = yes

	immediate = {
		add_deposit = d_ecological_disaster
		add_planet_devastation = 50
		if = {
			limit = {
				NOT = {
					has_deposit = d_darkling_rifts
				}
			}
			planet_event = { # Create Darkling Rifts if there aren't any, assumes first run.
				id = dark_world_events.5
				days = 10
			}
		}
	}

    option = {
        name = "Our time on this world is ending."
    }
}
