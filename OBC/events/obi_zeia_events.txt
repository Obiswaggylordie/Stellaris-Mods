# Existing Kvil'sha Leaders get traits
country_event = {
	id = zeia_events.02
	is_triggered_only = yes
	hide_window = yes
	
	trigger = {
		from = {
			species = { has_trait = trait_kvilsha }
			NOT = {
				has_trait = leader_trait_kvilsha
			}
		}
	}
	
	immediate = {
		from = {
			add_trait = {
                trait = leader_trait_kvilsha
                show_message = no
            }
		}
	}
}

# Guild Arch promotion
country_event = {
	id = zeia_events.04
	is_triggered_only = yes
	hide_window = yes
	
	trigger = {
		from = {
			species = { has_trait = trait_kvilsha }
			NOT = {
				has_trait = leader_trait_guild_grand
			}
		}
	}
	
	immediate = {
		from = {
            if = {
                limit = {
                    has_total_skill > 5
                }
                add_trait = {
                    trait = leader_trait_guild_grand
                    show_message = yes
                }
            }
		}
	}
}

# Auroras and Ascended Beings can't be killed
country_event = {
	id = zeia_events.07
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		fromfrom = {
			exists = leader
			fleet = { num_ships = 0 }
			leader = {
				OR = {
					has_trait = leader_trait_ascended_being
					has_trait = leader_trait_aurora_uviik
				}
			}
		}
	}

	immediate = {
		fromfrom = {
			leader = {
				exile_leader_as = obi_immortal_leader
			}
		}
		return_leader_from_exile = obi_immortal_leader
	}
}
