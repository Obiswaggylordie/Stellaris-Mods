celestial_energy_type_effect = {
	planet = {
		if = { # Gaia
			limit = {
				is_planet_class = pc_gaia
			}
			random_list = {
				33 = { set_planet_flag = ce_warm }
				33 = { set_planet_flag = ce_average }
				33 = { set_planet_flag = ce_cold }
			}
		}
		else_if = { # Tropical
			limit = {
				is_planet_class = pc_tropical
			}
			random_list = {
				33 = { set_planet_flag = ce_warm }
				33 = { set_planet_flag = ce_average }
				33 = { set_planet_flag = ce_cold }
			}
		}
		else_if = { # Continental
			limit = {
				is_planet_class = pc_continental
			}
			random_list = {
				50 = { set_planet_flag = ce_warm }
				50 = { set_planet_flag = ce_average }
			}
		}
		else_if = { # Ocean
			limit = {
				is_planet_class = pc_ocean
			}
			set_planet_flag = ce_cold
		}
		else_if = { # Arid
			limit = {
				is_planet_class = pc_arid
			}
			set_planet_flag = ce_warm
		}
		else_if = { # Alpine
			limit = {
				is_planet_class = pc_alpine
			}
			random_list = {
				50 = { set_planet_flag = ce_average }
				50 = { set_planet_flag = ce_cold }
			}
		}
		else_if = { # Savannah
			limit = {
				is_planet_class = pc_savannah
			}
			set_planet_flag = ce_warm
		}
		else_if = { # Tundra
			limit = {
				is_planet_class = pc_tundra
			}
			random_list = {
				50 = { set_planet_flag = ce_average }
				50 = { set_planet_flag = ce_cold }
			}
		}
		else_if = { # Desert
			limit = {
				is_planet_class = pc_desert
			}
			set_planet_flag = ce_warm
		}
		else_if = { # Arctic
			limit = {
				is_planet_class = pc_arctic
			}
			random_list = {
				50 = { set_planet_flag = ce_average }
				50 = { set_planet_flag = ce_cold }
			}
		}
		else_if = { # Barren
			limit = {
				is_planet_class = pc_barren
			}
			set_planet_flag = ce_warm
		}
		else_if = { # Barren Cold
			limit = {
				is_planet_class = pc_barren_cold
			}
			set_planet_flag = ce_average
		}
		else_if = { # Frozen
			limit = {
				is_planet_class = pc_frozen
			}
			set_planet_flag = ce_cold
		}
	}
}
